using System;
using System.Collections.Generic;

public static class RukbatText {
    private static List<int> Boundaries(string value) {
        var result = new List<int>();
        for (int index = 0; index < value.Length; index++) {
            result.Add(index);
            if (Char.IsHighSurrogate(value[index])) {
                if (index + 1 == value.Length || !Char.IsLowSurrogate(value[index + 1]))
                    throw new ArgumentException("Text contains an unpaired UTF-16 surrogate.");
                index++;
            } else if (Char.IsLowSurrogate(value[index])) {
                throw new ArgumentException("Text contains an unpaired UTF-16 surrogate.");
            }
        }
        result.Add(value.Length);
        return result;
    }
    public static long Length(string value) { return Boundaries(value).Count - 1; }
    public static string Slice(string value, long start, long count) {
        var positions = Boundaries(value);
        long length = positions.Count - 1;
        if (start < 0 || start > length || count < 0 || count > length - start)
            throw new ArgumentOutOfRangeException("range", "Text slice is outside the code-point range.");
        int first = positions[(int)start];
        return value.Substring(first, positions[(int)(start + count)] - first);
    }
    public static long? IndexOf(string value, string search) {
        var positions = Boundaries(value);
        Boundaries(search);
        int found = value.IndexOf(search, StringComparison.Ordinal);
        return found < 0 ? (long?)null : positions.BinarySearch(found);
    }
    public static string Upper(string value) { Boundaries(value); return value.ToUpperInvariant(); }
    public static string Lower(string value) { Boundaries(value); return value.ToLowerInvariant(); }
}
