using System;
using System.Collections.Generic;
using System.ComponentModel;
using System.IO;
using System.Runtime.InteropServices;
using System.Security.Cryptography;
using System.Text;
using Microsoft.Win32.SafeHandles;

public static class RukbatMoveBatch
{
    private const uint GenericRead = 0x80000000, GenericWrite = 0x40000000, DeleteAccess = 0x10000;
    private const uint ShareRead = 1, ShareWrite = 2, OpenExisting = 3, OpenAlways = 4;
    private const uint OpenReparsePoint = 0x200000, BackupSemantics = 0x2000000, WriteThrough = 0x80000000;
    private const uint DirectoryAttribute = 0x10, ReparseAttribute = 0x400;
    private const int FileRenameInformation = 3, JournalVersion = 1, MaximumPlanBytes = 64 * 1024 * 1024;
    private const int MaximumEntries = 100000, MaximumJournalOverhead = 1024;
    private static readonly byte[] Commit = Encoding.ASCII.GetBytes("COMMIT1\n");
    private static readonly Encoding JournalEncoding = new UTF8Encoding(false, true);
    private static readonly StringComparer Paths = StringComparer.OrdinalIgnoreCase;

    [StructLayout(LayoutKind.Sequential)] private struct FileInformation
    {
        public uint Attributes; public System.Runtime.InteropServices.ComTypes.FILETIME Creation, Access, Write;
        public uint Volume, SizeHigh, SizeLow, Links, IndexHigh, IndexLow;
    }
    [StructLayout(LayoutKind.Sequential, CharSet=CharSet.Unicode)] private struct RenameInformation
    {
        public uint Replace; public IntPtr Root; public uint Length; public char Name;
    }
    [DllImport("kernel32.dll", CharSet=CharSet.Unicode, SetLastError=true)] private static extern SafeFileHandle CreateFile(string path, uint access, uint share, IntPtr security, uint disposition, uint flags, IntPtr template);
    [DllImport("kernel32.dll", SetLastError=true)] private static extern bool GetFileInformationByHandle(SafeFileHandle handle, out FileInformation info);
    [DllImport("kernel32.dll", SetLastError=true)] private static extern bool SetFileInformationByHandle(SafeFileHandle handle, int kind, IntPtr info, uint size);

    private sealed class Entry
    {
        public string Source, Destination, Identity, Hash;
        public FileStream Stream;
        public bool Moved;
    }
    private sealed class Context : IDisposable
    {
        public readonly List<SafeFileHandle> Directories = new List<SafeFileHandle>();
        public readonly HashSet<string> LockedDirectories = new HashSet<string>(Paths);
        public readonly List<Entry> Entries = new List<Entry>();
        public void Dispose()
        {
            foreach (var entry in Entries) if (entry.Stream != null) entry.Stream.Dispose();
            foreach (var handle in Directories) handle.Dispose();
        }
    }
    private static IOException Error(string message) { return new IOException("files.move_batch: " + message); }
    private static FileInformation Information(SafeFileHandle handle)
    {
        FileInformation info;
        if (!GetFileInformationByHandle(handle, out info)) throw new Win32Exception(Marshal.GetLastWin32Error());
        if ((info.Attributes & ReparseAttribute) != 0) throw Error("Reparse points are not supported.");
        return info;
    }
    private static SafeFileHandle Open(string path, uint access, uint share, uint disposition, uint flags)
    {
        var handle = CreateFile(path, access, share, IntPtr.Zero, disposition, flags | OpenReparsePoint, IntPtr.Zero);
        if (handle.IsInvalid)
        {
            int code = Marshal.GetLastWin32Error();
            handle.Dispose();
            throw new Win32Exception(code, "Cannot lock path: " + path);
        }
        try { Information(handle); return handle; }
        catch { handle.Dispose(); throw; }
    }
    private static string Normalize(string value)
    {
        if (String.IsNullOrWhiteSpace(value)) throw Error("Empty path.");
        string path = Path.GetFullPath(value);
        JournalEncoding.GetByteCount(path);
        if (path.Length < 3 || path[1] != ':' || path[2] != '\\' || path.IndexOf(':', 2) >= 0)
            throw Error("Only local drive paths without alternate data streams are supported: " + value);
        foreach (var component in path.Substring(3).Split('\\'))
            if (component.Length == 0 || component.EndsWith(".") || component.EndsWith(" "))
                throw Error("Ambiguous path spelling: " + value);
        var drive = new DriveInfo(Path.GetPathRoot(path));
        if (drive.DriveType != DriveType.Fixed || drive.DriveFormat != "NTFS")
            throw Error("A local fixed NTFS volume is required: " + value);
        return path;
    }
    private static bool Exists(string path)
    {
        try { File.GetAttributes(path); return true; }
        catch (FileNotFoundException) { return false; }
        catch (DirectoryNotFoundException) { return false; }
    }
    private static void LockDirectory(Context context, string path, bool create)
    {
        if (context.LockedDirectories.Contains(path)) return;
        string parent = Path.GetDirectoryName(path);
        if (!String.IsNullOrEmpty(parent)) LockDirectory(context, parent, create);
        if (!Directory.Exists(path))
        {
            if (!create) throw Error("Missing directory: " + path);
            Directory.CreateDirectory(path);
        }
        var handle = Open(path, 0, ShareRead | ShareWrite, OpenExisting, BackupSemantics);
        context.Directories.Add(handle);
        if ((Information(handle).Attributes & DirectoryAttribute) == 0) throw Error("Expected directory: " + path);
        context.LockedDirectories.Add(path);
    }
    private static string Identity(FileInformation info)
    {
        return info.Volume.ToString("X8") + info.IndexHigh.ToString("X8") + info.IndexLow.ToString("X8") +
            ((uint)info.Creation.dwHighDateTime).ToString("X8") + ((uint)info.Creation.dwLowDateTime).ToString("X8");
    }
    private static string Digest(byte[] value)
    {
        using (var hash = SHA256.Create()) return Convert.ToBase64String(hash.ComputeHash(value));
    }
    private static void OpenEntry(Entry entry, string path)
    {
        var handle = Open(path, GenericRead | DeleteAccess, ShareRead, OpenExisting, 0);
        try
        {
            var info = Information(handle);
            if ((info.Attributes & DirectoryAttribute) != 0 || info.Links != 1)
                throw Error("Expected a regular file with one link: " + path);
            entry.Stream = new FileStream(handle, FileAccess.Read);
            string hash;
            using (var sha = SHA256.Create()) hash = Convert.ToBase64String(sha.ComputeHash(entry.Stream));
            string identity = Identity(info);
            if (entry.Identity != null && (entry.Identity != identity || entry.Hash != hash))
                throw Error("File identity or contents changed; recovery stopped: " + path);
            entry.Identity = identity; entry.Hash = hash;
        }
        catch { handle.Dispose(); throw; }
    }
    private static void Rename(Entry entry, string destination)
    {
        byte[] name = Encoding.Unicode.GetBytes(destination);
        int nameOffset = (int)Marshal.OffsetOf(typeof(RenameInformation), "Name");
        int size = nameOffset + name.Length + sizeof(char);
        IntPtr memory = Marshal.AllocHGlobal(size);
        try
        {
            for (int index = 0; index < size; index++) Marshal.WriteByte(memory, index, 0);
            Marshal.WriteInt32(memory, (int)Marshal.OffsetOf(typeof(RenameInformation), "Length"), name.Length);
            Marshal.Copy(name, 0, IntPtr.Add(memory, nameOffset), name.Length);
            // No replacement and no sharing of write/delete: the locked file itself is renamed.
            if (!SetFileInformationByHandle(entry.Stream.SafeFileHandle, FileRenameInformation, memory, (uint)size))
                throw new Win32Exception(Marshal.GetLastWin32Error(), "Cannot move to: " + destination);
        }
        finally { Marshal.FreeHGlobal(memory); }
    }
    private static byte[] Encode(List<Entry> entries)
    {
        using (var buffer = new MemoryStream())
        {
            using (var writer = new BinaryWriter(buffer, JournalEncoding, true))
            {
                writer.Write(JournalVersion); writer.Write(entries.Count);
                foreach (var entry in entries)
                {
                    writer.Write(entry.Source); writer.Write(entry.Destination);
                    writer.Write(entry.Identity); writer.Write(entry.Hash);
                }
            }
            return buffer.ToArray();
        }
    }
    private static bool Read(FileStream journal, Context context)
    {
        if (journal.Length > MaximumPlanBytes + MaximumJournalOverhead) throw Error("Journal exceeds the supported size.");
        using (var reader = new BinaryReader(journal, JournalEncoding, true))
        {
            int size = reader.ReadInt32();
            if (size <= 0 || size > MaximumPlanBytes) throw Error("Invalid journal length.");
            byte[] bytes = reader.ReadBytes(size);
            string hash = reader.ReadString();
            if (bytes.Length != size || hash != Digest(bytes)) throw Error("Journal checksum mismatch.");
            using (var payload = new MemoryStream(bytes))
            using (var data = new BinaryReader(payload, JournalEncoding))
            {
                if (data.ReadInt32() != JournalVersion) throw Error("Unsupported journal version.");
                int count = data.ReadInt32();
                if (count < 0 || count > MaximumEntries) throw Error("Invalid journal entry count.");
                for (int index = 0; index < count; index++)
                    context.Entries.Add(new Entry { Source = Normalize(data.ReadString()), Destination = Normalize(data.ReadString()), Identity = data.ReadString(), Hash = data.ReadString() });
                if (payload.Position != payload.Length) throw Error("Unexpected journal payload.");
            }
            long remaining = journal.Length - journal.Position;
            if (remaining == 0) return false;
            if (remaining != Commit.Length || Convert.ToBase64String(reader.ReadBytes(Commit.Length)) != Convert.ToBase64String(Commit))
                throw Error("Incomplete or invalid journal commit; no files changed.");
            return true;
        }
    }
    private static void Save(FileStream journal, List<Entry> entries)
    {
        byte[] bytes = Encode(entries);
        if (bytes.Length > MaximumPlanBytes) throw Error("Move plan exceeds 64 MiB.");
        using (var writer = new BinaryWriter(journal, JournalEncoding, true))
        {
            writer.Write(bytes.Length); writer.Write(bytes); writer.Write(Digest(bytes)); writer.Flush();
        }
        journal.Flush(true);
    }
    private static void ValidatePaths(Context context, string journalPath)
    {
        var paths = new HashSet<string>(Paths);
        paths.Add(journalPath);
        string root = Path.GetPathRoot(journalPath);
        foreach (var entry in context.Entries)
        {
            if (!Paths.Equals(root, Path.GetPathRoot(entry.Source)) || !Paths.Equals(root, Path.GetPathRoot(entry.Destination)))
                throw Error("Sources, destinations and journal must be on one NTFS volume.");
            if (!paths.Add(entry.Source) || !paths.Add(entry.Destination)) throw Error("Overlapping paths in move plan.");
        }
    }
    private static void OpenRecordedEntries(Context context, bool committed)
    {
        var identities = new HashSet<string>();
        foreach (var entry in context.Entries)
        {
            LockDirectory(context, Path.GetDirectoryName(entry.Source), false);
            LockDirectory(context, Path.GetDirectoryName(entry.Destination), false);
            bool source = Exists(entry.Source), destination = Exists(entry.Destination);
            if (source == destination || (committed && source))
                throw Error("Ambiguous or changed move state; no recovery performed: " + entry.Source);
            entry.Moved = destination;
            OpenEntry(entry, destination ? entry.Destination : entry.Source);
            if (!identities.Add(entry.Identity)) throw Error("Aliased files in journal.");
        }
    }
    private static void Rollback(Context context)
    {
        for (int index = context.Entries.Count - 1; index >= 0; index--)
        {
            Entry entry = context.Entries[index];
            if (!entry.Moved) continue;
            Rename(entry, entry.Source);
            entry.Moved = false;
        }
    }
    public static void Run(string[] sources, string[] destinations, string journalPath)
    {
        Execute(sources, destinations, journalPath, false);
    }
    public static void Recover(string journalPath)
    {
        Execute(null, null, journalPath, true);
    }
    private static void Execute(string[] sources, string[] destinations, string journalPath, bool recover)
    {
        if (!recover && (sources.Length != destinations.Length || sources.Length > MaximumEntries))
            throw Error("Move plans require matching arrays with at most 100000 entries.");
        journalPath = Normalize(journalPath);
        using (var context = new Context())
        {
            LockDirectory(context, Path.GetDirectoryName(journalPath), false);
            using (var handle = Open(journalPath, GenericRead | GenericWrite, 0, recover ? OpenExisting : OpenAlways, WriteThrough))
            using (var journal = new FileStream(handle, FileAccess.ReadWrite))
            {
                if (Information(handle).Links != 1) throw Error("Journal must not have hard links.");
                if (journal.Length != 0)
                {
                    bool committed = Read(journal, context);
                    ValidatePaths(context, journalPath);
                    if (!recover)
                    {
                        if (sources.Length != destinations.Length || sources.Length != context.Entries.Count) throw Error("Journal belongs to another plan.");
                        for (int index = 0; index < sources.Length; index++)
                            if (!Paths.Equals(Normalize(sources[index]), context.Entries[index].Source) || !Paths.Equals(Normalize(destinations[index]), context.Entries[index].Destination))
                                throw Error("Journal belongs to another plan.");
                    }
                    OpenRecordedEntries(context, committed);
                    if (committed) return;
                    Rollback(context);
                    if (recover) return;
                }
                else
                {
                    if (recover) throw Error("Empty journal; no recovery performed.");
                    if (sources.Length != destinations.Length) throw Error("Source and destination counts differ.");
                    for (int index = 0; index < sources.Length; index++)
                        context.Entries.Add(new Entry { Source = Normalize(sources[index]), Destination = Normalize(destinations[index]) });
                    ValidatePaths(context, journalPath);
                    var identities = new HashSet<string>();
                    foreach (var entry in context.Entries)
                    {
                        LockDirectory(context, Path.GetDirectoryName(entry.Source), false);
                        if (Exists(entry.Destination)) throw Error("Destination already exists: " + entry.Destination);
                        OpenEntry(entry, entry.Source);
                        if (!identities.Add(entry.Identity)) throw Error("Aliased source files.");
                    }
                    foreach (var entry in context.Entries) LockDirectory(context, Path.GetDirectoryName(entry.Destination), true);
                    Save(journal, context.Entries);
                }
                try
                {
                    foreach (var entry in context.Entries)
                    {
                        Rename(entry, entry.Destination);
                        entry.Moved = true;
                    }
                }
                catch (Exception failure)
                {
                    try { Rollback(context); }
                    catch (Exception rollback)
                    {
                        throw Error("Move failed: " + failure.Message + "; rollback incomplete: " + rollback.Message + "; retain journal for recovery: " + journalPath);
                    }
                    throw Error("Move failed and files were rolled back: " + failure.Message + "; journal: " + journalPath);
                }
                // A commit failure retains the completed files and journal for verified replay.
                journal.Write(Commit, 0, Commit.Length);
                journal.Flush(true);
            }
        }
    }
}
