using System;
using System.Collections.Generic;

public static class RukbatText {
    private static int NextIndex(string value, int index) {
        if (Char.IsHighSurrogate(value[index])) {
            if (index + 1 == value.Length || !Char.IsLowSurrogate(value[index + 1]))
                throw new ArgumentException("Text contains an unpaired UTF-16 surrogate.");
            return index + 2;
        }
        if (Char.IsLowSurrogate(value[index]))
            throw new ArgumentException("Text contains an unpaired UTF-16 surrogate.");
        return index + 1;
    }
    private static List<int> Boundaries(string value) {
        var result = new List<int>();
        for (int index = 0; index < value.Length; index = NextIndex(value, index)) {
            result.Add(index);
        }
        result.Add(value.Length);
        return result;
    }
    public static long Length(string value) {
        long count = 0;
        for (int index = 0; index < value.Length; index = NextIndex(value, index)) count++;
        return count;
    }
    public static string Slice(string value, long start, long count) {
        var positions = Boundaries(value);
        long length = positions.Count - 1;
        if (start < 0 || start > length || count < 0 || count > length - start)
            throw new ArgumentOutOfRangeException("range", "Text slice is outside the code-point range.");
        int first = positions[(int)start];
        return value.Substring(first, positions[(int)(start + count)] - first);
    }
    public static long? IndexOf(string value, string search) {
        var positions = Boundaries(value);
        Length(search);
        int found = value.IndexOf(search, StringComparison.Ordinal);
        return found < 0 ? (long?)null : positions.BinarySearch(found);
    }
    public static string Upper(string value) { Length(value); return value.ToUpperInvariant(); }
    public static string Lower(string value) { Length(value); return value.ToLowerInvariant(); }

    // Pass the existing argument array across the PowerShell boundary. Hosts that
    // format string arguments for invocation logging otherwise copy large inputs.
    public static long Length(object[] values) { return Length((string)values[0]); }
    public static string Slice(object[] values) {
        // Integer-producing APIs can box values as Int32 or Int64.
        return Slice((string)values[0], Convert.ToInt64(values[1]), Convert.ToInt64(values[2]));
    }
    public static long? IndexOf(object[] values) { return IndexOf((string)values[0], (string)values[1]); }
    public static string Upper(object[] values) { return Upper((string)values[0]); }
    public static string Lower(object[] values) { return Lower((string)values[0]); }
}
