using System;
using System.ComponentModel;
using System.Runtime.InteropServices;

// Windows conversions are used consistently on both .NET Framework and modern .NET.
public sealed class RukbatCodePageCodec {
    private const uint RejectInvalidBytes = 8;
    private const uint RejectInvalidUnicode = 128;
    private const uint DisableBestFit = 1024;
    private const uint Gb18030 = 54936;
    private const int InvalidSequence = 1113;
    // Encoding Standard, index gb18030 ranges: unmapped BMP gap and Unicode limit.
    private const int GbBmpGapStart = 39420;
    private const int GbSupplementaryStart = 189000;
    private const int GbMaximumPointer = 1237575;
    private readonly uint codePage;

    [DllImport("kernel32.dll")]
    [return: MarshalAs(UnmanagedType.Bool)]
    private static extern bool IsValidCodePage(uint codePage);
    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern int MultiByteToWideChar(uint codePage, uint flags,
        byte[] input, int byteCount, IntPtr output, int charCount);
    [DllImport("kernel32.dll", SetLastError = true, CharSet = CharSet.Unicode)]
    private static extern int WideCharToMultiByte(uint codePage, uint flags,
        string input, int charCount, byte[] output, int byteCount, IntPtr defaultChar, IntPtr usedDefaultChar);

    public RukbatCodePageCodec(int number) {
        // Stateful codecs and EUC-JP require separate strict validation policies.
        if (number <= 3 || number == 42 || number == 20932 || number == 51932 ||
            (number >= 50000 && number != Gb18030) || number == 1200 || number == 1201 ||
            number == 12000 || number == 12001 || !IsValidCodePage((uint)number))
            throw new ArgumentException("Unsupported strict Windows code page: " + number);
        codePage = (uint)number;
        // Validate both native flag contracts on the actual target before returning a value.
        GetString(new byte[] { 65 });
        GetBytes("A");
    }

    public string GetString(byte[] bytes) { return GetString(bytes, 0, bytes.Length); }
    public string GetString(byte[] bytes, int offset, int count) {
        if (offset < 0 || count < 0 || offset > bytes.Length - count) throw new ArgumentOutOfRangeException();
        if (count == 0) return String.Empty;
        if (codePage == Gb18030) ScanGb18030(bytes, offset, count, true);
        byte[] input = new byte[count];
        Buffer.BlockCopy(bytes, offset, input, 0, count);
        int length = MultiByteToWideChar(codePage, RejectInvalidBytes, input, count, IntPtr.Zero, 0);
        if (length == 0) throw new Win32Exception(Marshal.GetLastWin32Error(), "Invalid encoded input.");
        IntPtr buffer = Marshal.AllocHGlobal(checked(length * sizeof(char)));
        try {
            if (MultiByteToWideChar(codePage, RejectInvalidBytes, input, count, buffer, length) == 0)
                throw new Win32Exception(Marshal.GetLastWin32Error(), "Invalid encoded input.");
            return Marshal.PtrToStringUni(buffer, length);
        } finally { Marshal.FreeHGlobal(buffer); }
    }

    private static int ScanGb18030(byte[] bytes, int offset, int count, bool final) {
        int end = offset + count, i = offset;
        while (i < end) {
            int first = bytes[i];
            if (first <= 0x7f) { i++; continue; }
            if (first < 0x81 || first > 0xfe) throw new Win32Exception(InvalidSequence, "Invalid GB18030 input.");
            if (i + 1 == end) break;
            int second = bytes[i + 1];
            if ((second >= 0x40 && second <= 0x7e) || (second >= 0x80 && second <= 0xfe)) { i += 2; continue; }
            if (second < 0x30 || second > 0x39) throw new Win32Exception(InvalidSequence, "Invalid GB18030 input.");
            if (i + 2 == end) break;
            int third = bytes[i + 2];
            if (third < 0x81 || third > 0xfe) throw new Win32Exception(InvalidSequence, "Invalid GB18030 input.");
            if (i + 3 == end) break;
            int fourth = bytes[i + 3];
            if (fourth < 0x30 || fourth > 0x39) throw new Win32Exception(InvalidSequence, "Invalid GB18030 input.");
            int pointer = (((first - 0x81) * 10 + second - 0x30) * 126 + third - 0x81) * 10 + fourth - 0x30;
            if ((pointer >= GbBmpGapStart && pointer < GbSupplementaryStart) || pointer > GbMaximumPointer)
                throw new Win32Exception(InvalidSequence, "Unmapped GB18030 input.");
            i += 4;
        }
        if (final && i != end) throw new Win32Exception(InvalidSequence, "Incomplete GB18030 input.");
        return i - offset;
    }

    public Func<byte[], int, int, bool, string> CreateDecoder() {
        return new IncrementalDecoder(this).Convert;
    }

    private sealed class IncrementalDecoder {
        private const int MaximumSuffix = 3;
        private readonly RukbatCodePageCodec codec;
        private byte[] pending = new byte[0];
        private bool terminal;
        public IncrementalDecoder(RukbatCodePageCodec codec) { this.codec = codec; }
        public string Convert(byte[] bytes, int offset, int count, bool final) {
            if (terminal) throw new InvalidOperationException("Decoder has finished or failed.");
            if (offset < 0 || count < 0 || offset > bytes.Length - count) throw new ArgumentOutOfRangeException();
            try {
                byte[] block = new byte[checked(pending.Length + count)];
                Buffer.BlockCopy(pending, 0, block, 0, pending.Length);
                Buffer.BlockCopy(bytes, offset, block, pending.Length, count);
                int complete = block.Length;
                string result;
                if (codec.codePage == Gb18030) {
                    complete = ScanGb18030(block, 0, block.Length, final);
                    result = codec.GetString(block, 0, complete);
                } else if (final) {
                    result = codec.GetString(block);
                } else {
                    result = null;
                    for (int tail = 0; tail <= Math.Min(MaximumSuffix, block.Length); tail++) {
                        try { complete = block.Length - tail; result = codec.GetString(block, 0, complete); break; }
                        catch (Win32Exception error) { if (error.NativeErrorCode != InvalidSequence) throw; }
                    }
                    if (result == null) throw new Win32Exception(InvalidSequence, "Invalid encoded input.");
                }
                pending = new byte[block.Length - complete];
                Buffer.BlockCopy(block, complete, pending, 0, pending.Length);
                if (final) terminal = true;
                return result;
            } catch { pending = new byte[0]; terminal = true; throw; }
        }
    }

    public byte[] GetBytes(string text) {
        return GetBytes(text, Int32.MaxValue);
    }
    public byte[] GetBytes(string text, int maximum) {
        // Legacy conversion APIs do not reject malformed UTF-16 consistently.
        for (int i = 0; i < text.Length; i++) {
            if (Char.IsHighSurrogate(text[i])) {
                if (++i >= text.Length || !Char.IsLowSurrogate(text[i])) throw new ArgumentException("Invalid Unicode input.");
            } else if (Char.IsLowSurrogate(text[i])) throw new ArgumentException("Invalid Unicode input.");
        }
        if (text.Length == 0) return new byte[0];
        IntPtr used = codePage == Gb18030 ? IntPtr.Zero : Marshal.AllocHGlobal(sizeof(int));
        uint flags = codePage == Gb18030 ? RejectInvalidUnicode : DisableBestFit;
        try {
            if (used != IntPtr.Zero) Marshal.WriteInt32(used, 0);
            int length = WideCharToMultiByte(codePage, flags, text, text.Length, null, 0, IntPtr.Zero, used);
            if (length == 0) throw new Win32Exception(Marshal.GetLastWin32Error(), "Cannot encode Unicode input.");
            if (used != IntPtr.Zero && Marshal.ReadInt32(used) != 0) throw new ArgumentException("Unencodable Unicode input.");
            if (length > maximum) throw new ArgumentException("Encoded text exceeded its byte limit.");
            byte[] bytes = new byte[length];
            if (used != IntPtr.Zero) Marshal.WriteInt32(used, 0);
            if (WideCharToMultiByte(codePage, flags, text, text.Length, bytes, length, IntPtr.Zero, used) == 0)
                throw new Win32Exception(Marshal.GetLastWin32Error(), "Cannot encode Unicode input.");
            if (used != IntPtr.Zero && Marshal.ReadInt32(used) != 0) throw new ArgumentException("Unencodable Unicode input.");
            return bytes;
        } finally { if (used != IntPtr.Zero) Marshal.FreeHGlobal(used); }
    }
}
