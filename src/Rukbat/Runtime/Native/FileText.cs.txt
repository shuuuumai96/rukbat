using System;
using System.Collections.Generic;
using System.IO;
using System.Text;

public static class RukbatFileText {
    private const string Utf16LittleEndian = "utf-16le";
    private const string Utf16BigEndian = "utf-16be";
    private const string Utf32LittleEndian = "utf-32le";
    private const string Utf32BigEndian = "utf-32be";

    private struct Preamble {
        public readonly byte[] Bytes;
        public readonly string OppositeName;

        public Preamble(Encoding encoding, string oppositeName = "") {
            Bytes = encoding.GetPreamble();
            OppositeName = oppositeName;
        }
    }

    private static readonly Dictionary<string, Preamble> Preambles = new Dictionary<string, Preamble>(StringComparer.Ordinal) {
        { "utf-8", new Preamble(Encoding.UTF8) },
        { Utf16LittleEndian, new Preamble(Encoding.Unicode, Utf16BigEndian) },
        { Utf16BigEndian, new Preamble(Encoding.BigEndianUnicode, Utf16LittleEndian) },
        { Utf32LittleEndian, new Preamble(Encoding.UTF32, Utf32BigEndian) },
        { Utf32BigEndian, new Preamble(new UTF32Encoding(true, true), Utf32LittleEndian) }
    };

    public static int BomOffset(string name, byte[] bytes, int count) {
        Preamble preamble;
        if (name == null || !Preambles.TryGetValue(name, out preamble)) return 0;
        if (preamble.OppositeName.Length != 0 && MatchesPreamble(bytes, count, Preambles[preamble.OppositeName].Bytes))
            throw new ArgumentException("file encoding " + name + " conflicts with a " + preamble.OppositeName.ToUpperInvariant() + " BOM.");
        return MatchesPreamble(bytes, count, preamble.Bytes) ? preamble.Bytes.Length : 0;
    }

    private static bool MatchesPreamble(byte[] bytes, int count, byte[] preamble) {
        if (count < preamble.Length) return false;
        for (int i = 0; i < preamble.Length; i++) {
            if (bytes[i] != preamble[i]) return false;
        }
        return true;
    }

    public static Func<byte[], int, int, bool, string> CreateDecoder(Encoding encoding) {
        Decoder decoder = encoding.GetDecoder();
        bool terminal = false;
        return delegate(byte[] bytes, int offset, int count, bool final) {
            if (terminal) throw new InvalidOperationException("Decoder has finished or failed.");
            try {
                char[] chars = new char[encoding.GetMaxCharCount(count)];
                int length = decoder.GetChars(bytes, offset, count, chars, 0, final);
                if (final) terminal = true;
                return new string(chars, 0, length);
            } catch { terminal = true; throw; }
        };
    }

    public static RukbatTextFile Describe(string path, string encoding) { return new RukbatTextFile(path, encoding); }
    public static TextReader Open(RukbatTextFile source, Func<byte[], int, int, bool, string> decode) {
        Stream stream = File.Open(source.Path, FileMode.Open, FileAccess.Read, FileShare.Read);
        try { return new RukbatFileTextReader(stream, source.Encoding, decode); }
        catch { stream.Dispose(); throw; }
    }
}

public sealed class RukbatTextFile {
    public readonly string Path;
    public readonly string Encoding;
    internal RukbatTextFile(string path, string encoding) { Path = path; Encoding = encoding; }
}

// No IEnumerable: PowerShell must never enumerate a descriptor or cursor implicitly.
public sealed class RukbatFileLineCursor {
    private TextReader reader;
    private bool failed;
    public string Current { get; private set; }

    public RukbatFileLineCursor(TextReader reader) { this.reader = reader; }

    public bool MoveNext() {
        Current = null;
        if (reader == null || failed) throw new InvalidOperationException("Traversal has closed or failed.");
        try {
            Current = reader.ReadLine();
            return Current != null;
        } catch { failed = true; throw; }
    }

    public void Close() {
        TextReader owned = reader;
        reader = null;
        Current = null;
        if (owned != null) owned.Dispose();
    }
}

public sealed class RukbatFileTextReader : TextReader {
    private const int BufferSize = 64 * 1024;
    private const int BomProbeSize = 4;
    private Stream stream;
    private Func<byte[], int, int, bool, string> decode;
    private readonly string encoding;
    private byte[] buffer = new byte[BufferSize];
    private string chunk = String.Empty;
    private int position;
    private bool first = true, eof, skipLf;

    public RukbatFileTextReader(Stream stream, string encoding, Func<byte[], int, int, bool, string> decode) {
        if (stream == null || decode == null) throw new ArgumentNullException();
        this.stream = stream; this.encoding = encoding; this.decode = decode;
    }

    private bool HasCharacter() {
        if (stream == null) throw new ObjectDisposedException("RukbatFileTextReader");
        while (position == chunk.Length && !eof) ReadChunk();
        return position < chunk.Length;
    }

    public override int Read() { return HasCharacter() ? chunk[position++] : -1; }

    public override string ReadLine() {
        if (skipLf) {
            if (HasCharacter() && chunk[position] == '\n') position++;
            skipLf = false;
        }
        StringBuilder continued = null;
        while (HasCharacter()) {
            int start = position;
            while (position < chunk.Length && chunk[position] != '\r' && chunk[position] != '\n') position++;
            int length = position - start;
            if (position != chunk.Length) {
                skipLf = chunk[position++] == '\r';
                return continued == null ? chunk.Substring(start, length) : continued.Append(chunk, start, length).ToString();
            }
            if (continued == null) continued = new StringBuilder();
            continued.Append(chunk, start, length);
        }
        return continued == null ? null : continued.ToString();
    }

    private void ReadChunk() {
        int count, offset = 0;
        if (first) {
            count = 0;
            while (count < BomProbeSize) {
                int read = stream.Read(buffer, count, BomProbeSize - count);
                if (read == 0) { eof = true; break; }
                count += read;
            }
            offset = RukbatFileText.BomOffset(encoding, buffer, count);
            first = false;
        } else {
            count = stream.Read(buffer, 0, buffer.Length);
            if (count == 0) eof = true;
        }
        chunk = decode(buffer, offset, count - offset, eof);
        position = 0;
    }

    protected override void Dispose(bool disposing) {
        Stream owned = stream;
        stream = null; decode = null; buffer = null; chunk = null;
        if (disposing && owned != null) owned.Dispose();
    }
}
